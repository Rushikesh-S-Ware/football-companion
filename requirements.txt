# Football Companion Agent — Python dependencies
# Install into a virtual environment:  pip install -r requirements.txt
#
# Versions use ">=" (a floor, not a pin) so a clean install on Python 3.12
# picks up compatible releases. If a future upgrade ever breaks us, we can pin.

# --- Data fetching ---
requests>=2.31          # simple HTTP client for the football APIs
feedparser>=6.0         # parses RSS news feeds (Guardian, ESPN, etc.)

# --- Storage ---
duckdb>=1.1             # single-file analytical database for timestamped snapshots

# --- The agent ---
groq>=0.13              # Groq SDK — fast Llama inference with OpenAI-style tool calling
python-dotenv>=1.0      # loads API keys from the .env file

# --- Data handling / UI ---
pandas>=2.2             # tabular data wrangling between DuckDB and Streamlit
streamlit>=1.40         # one-page dashboard UI (Phase 4)

# --- Tests ---
pytest>=8.0             # tests for the deterministic parts (data parsing, memory I/O)
